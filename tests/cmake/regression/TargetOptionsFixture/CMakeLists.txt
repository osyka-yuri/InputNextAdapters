get_property(_cached_release_c_flags_before CACHE CMAKE_C_FLAGS_RELEASE PROPERTY VALUE)
get_property(_cached_release_cxx_flags_before CACHE CMAKE_CXX_FLAGS_RELEASE PROPERTY VALUE)
set(_directory_release_c_flags_before "${CMAKE_C_FLAGS_RELEASE}")
set(_directory_release_cxx_flags_before "${CMAKE_CXX_FLAGS_RELEASE}")
function(_check_production_release_normalization out_c_flags out_cxx_flags)
    set(CMAKE_C_FLAGS_RELEASE "/O2 /Ob2 /DNDEBUG")
    set(CMAKE_CXX_FLAGS_RELEASE "/O2 /Ob2 /DNDEBUG")
    inputnext_prepare_production_configuration()
    foreach(_language IN ITEMS C CXX)
        if(CMAKE_${_language}_FLAGS_RELEASE MATCHES "(^|[ \t])[/-]Ob[0-3]([ \t]|$)")
            message(FATAL_ERROR "Production left the default /Ob2 on ${_language}_FLAGS_RELEASE.")
        endif()
        if(NOT CMAKE_${_language}_FLAGS_RELEASE MATCHES "(^|[ \t])/O2([ \t]|$)")
            message(FATAL_ERROR "Production removed /O2 from ${_language}_FLAGS_RELEASE.")
        endif()
    endforeach()
    set(${out_c_flags} "${CMAKE_C_FLAGS_RELEASE}" PARENT_SCOPE)
    set(${out_cxx_flags} "${CMAKE_CXX_FLAGS_RELEASE}" PARENT_SCOPE)
endfunction()
_check_production_release_normalization(_normalized_c_flags _normalized_cxx_flags)
if(NOT _normalized_c_flags STREQUAL "/O2 /DNDEBUG" OR
   NOT _normalized_cxx_flags STREQUAL "/O2 /DNDEBUG")
    message(FATAL_ERROR "Production did not normalize only the inherited inline option.")
endif()
if(NOT CMAKE_C_FLAGS_RELEASE STREQUAL _directory_release_c_flags_before OR
   NOT CMAKE_CXX_FLAGS_RELEASE STREQUAL _directory_release_cxx_flags_before)
    message(FATAL_ERROR "Production normalization escaped its caller scope.")
endif()
get_property(_cached_release_c_flags_after CACHE CMAKE_C_FLAGS_RELEASE PROPERTY VALUE)
get_property(_cached_release_cxx_flags_after CACHE CMAKE_CXX_FLAGS_RELEASE PROPERTY VALUE)
if(NOT _cached_release_c_flags_after STREQUAL _cached_release_c_flags_before OR
   NOT _cached_release_cxx_flags_after STREQUAL _cached_release_cxx_flags_before)
    message(FATAL_ERROR "Production normalization modified CMake cache flags.")
endif()
function(_check_diagnostic_release_flags)
    set(INPUTNEXTADAPTERS_ENABLE_MSVC_ANALYZE ON)
    set(CMAKE_CXX_FLAGS_RELEASE "/O2 /Ob2 /DNDEBUG")
    inputnext_prepare_production_configuration()
    if(NOT CMAKE_CXX_FLAGS_RELEASE MATCHES "(^|[ \t])[/-]Ob2([ \t]|$)")
        message(FATAL_ERROR "Production normalization must leave diagnostic-profile flags alone.")
    endif()
endfunction()
_check_diagnostic_release_flags()

function(_assert_target_option target property expected)
    get_target_property(_actual "${target}" "${property}")
    if(NOT "${_actual}" MATCHES "${expected}")
        message(FATAL_ERROR "${target}.${property} lacks '${expected}'; got '${_actual}'.")
    endif()
endfunction()

function(_assert_target_option_absent target property unexpected)
    get_target_property(_actual "${target}" "${property}")
    if("${_actual}" MATCHES "${unexpected}")
        message(FATAL_ERROR "${target}.${property} unexpectedly contains '${unexpected}'; got '${_actual}'.")
    endif()
endfunction()

add_library(product_static STATIC "${_source}")
inputnext_configure_target(product_static PRODUCTION)
get_target_property(_product_static_ipo product_static INTERPROCEDURAL_OPTIMIZATION_RELEASE)
if(NOT _product_static_ipo)
    message(FATAL_ERROR "Production static targets must enable Release IPO/LTCG.")
endif()
foreach(_flag IN ITEMS /sdl /options:strict /w44242 /w44254 /w44287 /w44296 /w44062 /w44265 /w44365 /w44388)
    _assert_target_option(product_static COMPILE_OPTIONS "${_flag}")
endforeach()
foreach(_flag IN ITEMS /O2 /Ob3 /Gw /Gy /Zi /fp:precise)
    _assert_target_option(product_static COMPILE_OPTIONS "${_flag}")
endforeach()
_assert_target_option(product_static COMPILE_OPTIONS "CONFIG:Release.*COMPILE_LANG_AND_ID:CXX,MSVC.*:/O2")
_assert_target_option_absent(product_static INTERFACE_COMPILE_OPTIONS "/W4")
_assert_target_option_absent(product_static INTERFACE_COMPILE_OPTIONS "/Ob3")
get_target_property(_product_static_crt product_static MSVC_RUNTIME_LIBRARY)
if(_product_static_crt AND NOT _product_static_crt MATCHES "-NOTFOUND$")
    message(FATAL_ERROR "Owned targets must not set a consumer CRT: '${_product_static_crt}'.")
endif()
